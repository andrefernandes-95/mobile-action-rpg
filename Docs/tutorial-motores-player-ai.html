Tutorial completo · versão separada

Um manager agnóstico.
Dois motores simples.

PlayerMotor trata apenas do CharacterController. AIMotor trata apenas do NavMeshAgent. O CharacterManager não sabe qual dos dois está activo: conhece apenas um contrato mínimo.

01 · Objectivo

Cada classe tem um trabalho óbvio

Player

Recebe uma direcção do joystick ou teclado e move imediatamente o CharacterController.

AI

Recebe um ponto no mundo, entrega-o ao NavMeshAgent e pára quando entra em combate.

O manager continua responsável por animação, rotação, estado ocupado e referências de gameplay. Não escolhe caminhos, não lê input e não conhece NavMeshAgent nem CharacterController.

Regra principal: os sistemas que sabem o que querem movem o motor certo. PlayerInput usa PlayerMotor; a máquina de estados usa AIMotor.

04 · Movimento do Player

PlayerMotor.cs

O Player não tem destino, path ou stopping distance. O stick dá uma direcção; o controller move-se nesse frame.

Assets/Scripts/Movement/PlayerMotor.cs
using UnityEngine;

namespace AF
{
    [RequireComponent(typeof(CharacterController))]
    public sealed class PlayerMotor : MonoBehaviour, IMovementMotor
    {
        [SerializeField] float runSpeed = 5.5f;
        [SerializeField] float gravity = -20f;

        CharacterController controller;
        Vector3 velocity;

        public float MoveAmount { get; private set; }
        public float RunSpeed => runSpeed;
        public Vector3 PlanarVelocity => new Vector3(velocity.x, 0f, velocity.z);
        public bool IsMotorEnabled => controller != null && controller.enabled;

        void Awake()
        {
            controller = GetComponent<CharacterController>();
            controller.enabled = false;
        }

        public void Move(Vector3 direction, float moveAmount)
        {
            if (!IsMotorEnabled)
            {
                return;
            }

            MoveAmount = Mathf.Clamp01(moveAmount);
            direction.y = 0f;

            if (direction.sqrMagnitude < 0.0001f || MoveAmount <= 0f)
            {
                Stop();
                return;
            }

            direction.Normalize();
            velocity.x = direction.x * runSpeed * MoveAmount;
            velocity.z = direction.z * runSpeed * MoveAmount;

            ApplyGravity();
            controller.Move(velocity * Time.deltaTime);
        }

        public void Stop()
        {
            MoveAmount = 0f;
            velocity.x = 0f;
            velocity.z = 0f;

            if (!IsMotorEnabled)
            {
                return;
            }

            ApplyGravity();
            controller.Move(Vector3.up * velocity.y * Time.deltaTime);
        }

        void ApplyGravity()
        {
            if (controller.isGrounded && velocity.y < 0f)
            {
                velocity.y = -2f;
            }
            else
            {
                velocity.y += gravity * Time.deltaTime;
            }
        }

        public void ApplyDisplacement(Vector3 worldDelta)
        {
            if (IsMotorEnabled)
            {
                controller.Move(worldDelta);
            }
        }

        public void SetMotorEnabled(bool enabled)
        {
            controller.enabled = enabled;

            if (!enabled)
            {
                velocity = Vector3.zero;
                MoveAmount = 0f;
            }
        }
    }
}
05 · Movimento da AI

AIMotor.cs

A AI só recebe destinos. O NavMeshAgent calcula o caminho, a velocidade e a distância restante.

Assets/Scripts/Movement/AIMotor.cs
using UnityEngine;
using UnityEngine.AI;

namespace AF
{
    [RequireComponent(typeof(NavMeshAgent))]
    public sealed class AIMotor : MonoBehaviour, IMovementMotor
    {
        [SerializeField] float runSpeed = 5.5f;

        NavMeshAgent agent;

        public float MoveAmount { get; private set; }
        public float RunSpeed => runSpeed;
        public float StoppingDistance => agent.stoppingDistance;
        public float RemainingDistance => IsMotorEnabled ? agent.remainingDistance : 0f;
        public bool HasPath => IsMotorEnabled && agent.hasPath;
        public bool IsMotorEnabled => agent != null && agent.enabled && agent.isOnNavMesh;

        public Vector3 PlanarVelocity
        {
            get
            {
                if (!IsMotorEnabled)
                {
                    return Vector3.zero;
                }

                Vector3 velocity = agent.velocity;
                velocity.y = 0f;
                return velocity;
            }
        }

        void Awake()
        {
            agent = GetComponent<NavMeshAgent>();
            agent.updateRotation = false;

            if (runSpeed <= 0f)
            {
                runSpeed = agent.speed;
            }

            agent.enabled = false;
        }

        public void SetDestination(Vector3 worldPosition)
        {
            if (!IsMotorEnabled)
            {
                return;
            }

            MoveAmount = 1f;
            agent.speed = runSpeed;
            agent.SetDestination(worldPosition);
        }

        public void Stop()
        {
            MoveAmount = 0f;

            if (!IsMotorEnabled)
            {
                return;
            }

            agent.ResetPath();
            agent.velocity = Vector3.zero;
        }

        public void ApplyDisplacement(Vector3 worldDelta)
        {
            if (IsMotorEnabled)
            {
                agent.Move(worldDelta);
            }
        }

        public void SetMotorEnabled(bool enabled)
        {
            if (!enabled)
            {
                Stop();
            }

            agent.enabled = enabled;
        }
    }
}
Importante: o NavMeshAgent só pode ser activado sobre um NavMesh válido. Confirma que os prefabs nascem em cima da área navegável.

CharacterManager.cs completo

Repara que esta classe não contém os nomes PlayerMotor, AIMotor, CharacterController ou NavMeshAgent.

Assets/Scripts/Core/CharacterManager.cs
namespace AF
{
    using UnityEngine;

    public class CharacterManager : MonoBehaviour
    {
        [Header("Name")]
        public new string name = "";

        [Header("Components")]
        public Animator animator;

        [Header("Character Components")]
        public Health health;
        public CharacterDamageReceiver characterDamageReceiver;
        public Dodge dodge;
        public Perception perception;
        public LockOn lockOn;
        public CombatManager combatManager;

        [Header("Locomotion")]
        public float rotationSpeed = 10f;
        public float animSpeedDamp = 0.12f;

        [Header("Patrol")]
        public Transform[] patrolPoints;

        public bool isBusy = false;

        IMovementMotor motor;
        float inputMoveAmount;
        Vector3 inputMoveDirection;
        Vector3 defaultPosition;

        public IMovementMotor Motor => motor;
        public float MoveAmount => motor != null ? motor.MoveAmount : 0f;

        void Awake()
        {
            defaultPosition = transform.position;
        }

        void Update()
        {
            UpdateRotation();
            UpdateAnimation();
        }

        public void SetMotor(IMovementMotor newMotor)
        {
            if (motor == newMotor)
            {
                return;
            }

            if (motor != null)
            {
                motor.SetMotorEnabled(false);
            }

            motor = newMotor;

            if (motor != null)
            {
                motor.SetMotorEnabled(true);
            }
        }

        public void SetMovementInput(Vector3 direction, float moveAmount)
        {
            inputMoveDirection = direction;
            inputMoveAmount = Mathf.Clamp01(moveAmount);
        }

        public void OnIdle()
        {
            isBusy = false;
        }

        public void Stop()
        {
            inputMoveAmount = 0f;
            inputMoveDirection = Vector3.zero;

            if (motor != null)
            {
                motor.Stop();
            }
        }

        public void ApplyDisplacement(Vector3 worldDelta)
        {
            if (motor != null)
            {
                motor.ApplyDisplacement(worldDelta);
            }
        }

        void UpdateRotation()
        {
            if (health != null && health.IsDead)
            {
                return;
            }

            if (motor == null)
            {
                return;
            }

            if (lockOn != null && lockOn.isLockedOn && lockOn.lockOnTarget != null)
            {
                Vector3 direction = lockOn.lockOnTarget.position - transform.position;
                direction.y = 0f;

                if (direction.sqrMagnitude < 0.0001f)
                {
                    return;
                }

                Quaternion targetRotation = Quaternion.LookRotation(direction.normalized);
                float amount = Mathf.Clamp01(lockOn.lockOnRotationSpeed * Time.deltaTime);
                transform.rotation = Quaternion.Slerp(transform.rotation, targetRotation, amount);
                return;
            }

            Vector3 moveDirection = motor.PlanarVelocity;
            float minimum = IsPlayer() ? 0.01f : 0.08f;

            if (moveDirection.sqrMagnitude < minimum)
            {
                return;
            }

            Quaternion moveRotation = Quaternion.LookRotation(moveDirection.normalized);
            float speed = rotationSpeed * Mathf.Lerp(0.35f, 1f, MoveAmount);
            transform.rotation = Quaternion.Slerp(
                transform.rotation,
                moveRotation,
                speed * Time.deltaTime
            );
        }

        void UpdateAnimation()
        {
            if (animator == null || motor == null)
            {
                return;
            }

            float vertical = 0f;
            float horizontal = 0f;
            bool lockedOn = lockOn != null && lockOn.isLockedOn && lockOn.lockOnTarget != null;

            if (IsPlayer() && lockedOn)
            {
                if (inputMoveAmount > 0.01f && inputMoveDirection.sqrMagnitude > 0.0001f)
                {
                    Vector3 toTarget = lockOn.lockOnTarget.position - transform.position;
                    toTarget.y = 0f;

                    if (toTarget.sqrMagnitude > 0.0001f)
                    {
                        Quaternion faceRotation = Quaternion.LookRotation(toTarget.normalized);
                        Vector3 localDirection = Quaternion.Inverse(faceRotation) * inputMoveDirection;
                        localDirection.y = 0f;

                        float largestAxis = Mathf.Max(
                            Mathf.Abs(localDirection.x),
                            Mathf.Abs(localDirection.z)
                        );

                        if (largestAxis > 0.0001f)
                        {
                            vertical = localDirection.z / largestAxis * inputMoveAmount;
                            horizontal = localDirection.x / largestAxis * inputMoveAmount;
                        }
                    }
                }

                animator.SetFloat(AnimHashes.Vertical, vertical);
                animator.SetFloat(AnimHashes.Horizontal, horizontal);
                return;
            }

            if (IsPlayer())
            {
                vertical = inputMoveAmount;
            }
            else
            {
                float runSpeed = Mathf.Max(0.01f, motor.RunSpeed);
                vertical = Mathf.Clamp01(motor.PlanarVelocity.magnitude / runSpeed);
            }

            animator.SetFloat(AnimHashes.Vertical, vertical, animSpeedDamp, Time.deltaTime);
            animator.SetFloat(AnimHashes.Horizontal, 0f, animSpeedDamp, Time.deltaTime);
        }

        public bool IsPlayer()
        {
            return GameContext.Player == this;
        }

        public CharacterManager GetPlayer()
        {
            return GameContext.Player;
        }

        public void RestoreDefaultPosition()
        {
            transform.position = defaultPosition;
        }
    }
}
07 · Controlo humano

PlayerInput.cs completo

A troca de personagem actualiza primeiro o GameContext.Player. Depois devolve a personagem anterior à AI e activa o PlayerMotor na nova.

Assets/Scripts/Core/PlayerInput.cs
namespace AF
{
    using UnityEngine;
    using UnityEngine.Events;

    public class PlayerInput : MonoBehaviour
    {
        Camera mainCam;
        PlayerMotor playerMotor;

        public Joystick joystick;
        public CharacterManager startingCharacterManager;

        [Header("Current Character")]
        public CharacterManager characterManager;

        [HideInInspector] public UnityEvent<CharacterManager> onSetCurrentCharacter;

        public float MoveAmount { get; private set; }
        public Vector3 MoveDirection { get; private set; }

        void Start()
        {
            mainCam = Camera.main;
            SetCharacter(startingCharacterManager);
        }

        void SetCharacter(CharacterManager character)
        {
            if (character == null)
            {
                return;
            }

            CharacterManager previousCharacter = characterManager;
            GameContext.SetPlayer(character);

            if (previousCharacter != null && previousCharacter != character)
            {
                StateMachine previousAI = previousCharacter.GetComponent<StateMachine>();

                if (previousAI != null)
                {
                    previousAI.EnableAI();
                }
            }

            characterManager = character;
            playerMotor = characterManager.GetComponent<PlayerMotor>();

            if (playerMotor == null)
            {
                Debug.LogError($"{characterManager.name}: falta PlayerMotor.", characterManager);
                return;
            }

            characterManager.SetMotor(playerMotor);
            onSetCurrentCharacter?.Invoke(characterManager);
        }

        void Update()
        {
            if (Input.GetKeyDown(KeyCode.Space))
            {
                OnSpecialAbility();
                return;
            }

            HandleMovement();
        }

        void HandleMovement()
        {
            MoveAmount = 0f;
            MoveDirection = Vector3.zero;

            if (mainCam == null || characterManager == null || playerMotor == null)
            {
                return;
            }

            if (characterManager.isBusy)
            {
                return;
            }

            float horizontal = Input.GetAxis("Horizontal");
            float vertical = Input.GetAxis("Vertical");

            if (joystick != null)
            {
                horizontal += joystick.Horizontal;
                vertical += joystick.Vertical;
            }

            Vector3 cameraForward = mainCam.transform.forward;
            Vector3 cameraRight = mainCam.transform.right;
            cameraForward.y = 0f;
            cameraRight.y = 0f;
            cameraForward.Normalize();
            cameraRight.Normalize();

            Vector3 movement = cameraForward * vertical + cameraRight * horizontal;
            float amount = Mathf.Clamp01(movement.magnitude);

            if (amount < 0.01f)
            {
                characterManager.SetMovementInput(Vector3.zero, 0f);
                playerMotor.Stop();
                return;
            }

            MoveAmount = amount;
            MoveDirection = movement / amount;
            characterManager.SetMovementInput(MoveDirection, MoveAmount);
            playerMotor.Move(MoveDirection, MoveAmount);
        }

        public void OnAttack()
        {
            if (characterManager != null)
            {
                characterManager.combatManager.Attack();
            }
        }

        public void OnSpecialAbility()
        {
            if (characterManager != null)
            {
                characterManager.combatManager.OnSpecialAbility();
            }
        }

        public Vector3 GetDodgeDirection(bool isLockedOn)
        {
            float horizontal = Input.GetAxis("Horizontal");
            float vertical = Input.GetAxis("Vertical");

            if (joystick != null)
            {
                horizontal += joystick.Horizontal;
                vertical += joystick.Vertical;
            }

            float forwardDirection = isLockedOn ? -1f : 1f;
            Vector3 cameraForward = mainCam.transform.forward * forwardDirection;
            Vector3 cameraRight = mainCam.transform.right;
            cameraForward.y = 0f;
            cameraRight.y = 0f;
            cameraForward.Normalize();
            cameraRight.Normalize();

            Vector3 dodgeDirection = cameraForward * vertical + cameraRight * horizontal;

            if (dodgeDirection.sqrMagnitude < 0.01f)
            {
                return characterManager.transform.forward;
            }

            return dodgeDirection.normalized;
        }
    }
}
08 · Cérebro da AI

StateMachine.cs completo

A máquina de estados é dona do AIMotor. Quando uma personagem deixa de ser Player, EnableAI() injecta esse motor no manager.

Assets/Scripts/AI/StateMachine.cs
namespace AF
{
    using UnityEngine;

    public class StateMachine : MonoBehaviour
    {
        public CharacterManager character;

        [Header("AI States")]
        public AIState idleState;
        public AIState patrolState;
        public AIState chaseState;
        public AIState combatState;

        [HideInInspector] public AIState currentState;

        public readonly CombatStateRuntime CombatRuntime = new();
        public AIMotor Motor { get; private set; }

        void Awake()
        {
            if (character == null)
            {
                character = GetComponent<CharacterManager>();
            }

            Motor = GetComponent<AIMotor>();
        }

        void Start()
        {
            if (!character.IsPlayer())
            {
                EnableAI();
            }
        }

        void Update()
        {
            if (!character.IsPlayer() && Motor != null && Motor.IsMotorEnabled)
            {
                currentState?.Tick(this);
            }
        }

        public void EnableAI()
        {
            if (Motor == null)
            {
                Debug.LogError($"{character.name}: falta AIMotor.", character);
                return;
            }

            character.SetMotor(Motor);

            if (currentState == null)
            {
                SwitchState(patrolState != null ? patrolState : idleState);
            }
        }

        public void SwitchState(AIState newState)
        {
            if (newState == null || currentState == newState)
            {
                return;
            }

            currentState?.Exit(this);
            currentState = newState;
            currentState.Enter(this);
        }
    }
}

namespace AF
{
    using UnityEngine;

    [CreateAssetMenu(menuName = "AF/AI/States/Patrol")]
    public class StatePatrol : AIState
    {
        int patrolIndex;

        public override void Enter(StateMachine controller)
        {
            patrolIndex = 0;
        }

        public override void Tick(StateMachine controller)
        {
            if (controller.character.perception.CanSeePlayer())
            {
                controller.SwitchState(controller.chaseState);
                return;
            }

            Transform[] points = controller.character.patrolPoints;

            if (points == null || points.Length == 0)
            {
                return;
            }

            bool needsNewPoint = !controller.Motor.HasPath
                || controller.Motor.RemainingDistance < controller.Motor.StoppingDistance;

            if (needsNewPoint)
            {
                controller.Motor.SetDestination(points[patrolIndex].position);
                patrolIndex = (patrolIndex + 1) % points.Length;
            }
        }

        public override void Exit(StateMachine controller)
        {
            controller.Motor.Stop();
        }
    }
}

namespace AF
{
    using UnityEngine;

    [CreateAssetMenu(menuName = "AF/AI/States/Chase")]
    public class StateChase : AIState
    {
        public float disengageMultiplier = 1.3f;

        public override void Enter(StateMachine controller)
        {
        }

        public override void Tick(StateMachine controller)
        {
            CharacterManager player = controller.character.GetPlayer();

            if (player == null || controller.character.isBusy)
            {
                return;
            }

            if (controller.character.health.IsDead || player.health.IsDead)
            {
                controller.SwitchState(
                    controller.patrolState != null
                        ? controller.patrolState
                        : controller.idleState
                );
                return;
            }

            float distance = Vector3.Distance(
                controller.transform.position,
                player.transform.position
            );

            if (distance <= controller.Motor.StoppingDistance)
            {
                controller.SwitchState(controller.combatState);
                return;
            }

            controller.Motor.SetDestination(player.transform.position);
        }

        public override void Exit(StateMachine controller)
        {
            controller.Motor.Stop();
        }
    }
}

A paragem que pediste está no Enter: assim que a AI entra em combate, o path é apagado e a velocidade do agent fica a zero.
Assets/Scripts/AI/StateCombat.cs
namespace AF
{
    using UnityEngine;

    [CreateAssetMenu(menuName = "AF/AI/States/Combat")]
    public class StateCombat : AIState
    {
        [Header("Attack")]
        public float attackCooldown = 1.2f;

        [Header("Disengage")]
        public float disengageMultiplier = 1.3f;

        public override void Enter(StateMachine controller)
        {
            controller.Motor.Stop();
        }

        public override void Tick(StateMachine controller)
        {
            CharacterManager player = controller.character.GetPlayer();

            if (player == null)
            {
                return;
            }

            if (player.health.IsDead)
            {
                controller.SwitchState(
                    controller.patrolState != null
                        ? controller.patrolState
                        : controller.idleState
                );
                return;
            }

            float distance = Vector3.Distance(
                controller.transform.position,
                player.transform.position
            );

            float sight = controller.character.perception.sightRange;
            float disengageDistance = sight * disengageMultiplier;

            if (distance > disengageDistance
                || distance > controller.Motor.StoppingDistance)
            {
                controller.SwitchState(controller.chaseState);
                return;
            }

            Vector3 direction = player.transform.position - controller.transform.position;
            direction.y = 0f;

            if (direction.sqrMagnitude > 0.01f)
            {
                Quaternion rotation = Quaternion.LookRotation(direction);
                controller.transform.rotation = Quaternion.Slerp(
                    controller.transform.rotation,
                    rotation,
                    Time.deltaTime * controller.character.rotationSpeed
                );
            }

            bool cooldownFinished = Time.time
                >= controller.CombatRuntime.LastAttackTime + attackCooldown;

            if (!controller.character.isBusy && cooldownFinished)
            {
                controller.CombatRuntime.LastAttackTime = Time.time;

                if (Random.Range(0f, 1f) >= 0.15f)
                {
                    controller.character.combatManager.Attack();
                }
            }
        }

        public override void Exit(StateMachine controller)
        {
        }
    }
}
10 · Pequena adaptação

CombatManager.cs completo

A única alteração é o fallback de GetEngageRadius(): quando não há uma arma válida, lê a distância no AIMotor em vez de a procurar na interface comum.

Assets/Scripts/Combat/CombatManager.cs
using UnityEngine;
using UnityEngine.Events;

namespace AF
{
    public class CombatManager : MonoBehaviour
    {
        [Header("Abilities")]
        [SerializeField] Ability specialAbility;
        public Ability SpecialAbility => specialAbility;

        [Header("Components")]
        [SerializeField] CharacterManager characterManager;
        [SerializeField] EquipmentManager equipmentManager;

        [Header("Debug")]
        [SerializeField] Ability currentAbility;

        [HideInInspector] public UnityEvent<Hitbox> onHitboxOpen;
        [HideInInspector] public UnityEvent<Hitbox> onHitboxClose;
        [HideInInspector] public UnityEvent<Vector3> onEnemyHit;

        float abilityCooldownEndTime;

        public void ResetStates()
        {
        }

        public void Attack()
        {
            if (characterManager.isBusy)
            {
                return;
            }

            if (Time.time < abilityCooldownEndTime)
            {
                return;
            }

            if (equipmentManager == null || equipmentManager.weaponInstance == null)
            {
                return;
            }

            WeaponAbility weaponAbility = equipmentManager.weaponInstance.ability;

            if (weaponAbility == null)
            {
                return;
            }

            characterManager.Stop();
            SetCurrentAbility(weaponAbility, true);
        }

        public void OnSpecialAbility()
        {
            if (specialAbility != null)
            {
                SetCurrentAbility(specialAbility, false);
            }
        }

        public void SetCurrentAbility(Ability next, bool updateCooldown)
        {
            if (currentAbility != null)
            {
                currentAbility.OnEnd(characterManager);
            }

            currentAbility = next;
            currentAbility?.OnStart(characterManager);

            if (updateCooldown && currentAbility != null)
            {
                abilityCooldownEndTime = Time.time + currentAbility.cooldown;
            }
        }

        public bool TryBuildDamagePacket(Vector3 hitPoint, out DamagePacket packet)
        {
            if (currentAbility == null)
            {
                packet = default;
                return false;
            }

            return currentAbility.TryBuildDamagePacket(characterManager, hitPoint, out packet);
        }

        public float GetEngageRadius()
        {
            if (TryGetCurrentWeaponAbility(out WeaponAbility weaponAbility))
            {
                return weaponAbility.engageRadius;
            }

            AIMotor aiMotor = GetComponent<AIMotor>();

            if (aiMotor != null)
            {
                return aiMotor.StoppingDistance;
            }

            return 1.5f;
        }

        bool TryGetCurrentWeaponAbility(out WeaponAbility weaponAbility)
        {
            weaponAbility = null;

            if (equipmentManager == null || equipmentManager.weaponInstance == null)
            {
                return false;
            }

            weaponAbility = equipmentManager.weaponInstance.ability;
            return weaponAbility != null;
        }
    }
}
